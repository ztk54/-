## 归并排序
归并排序 = 分治法，核心三步：
1. **分割**：二分数组，直到每个子数组长度为 1。
2. **递归排序**：左右子数组各自有序。
3. **合并**：合并两个有序子数组为一个有序数组。
时间复杂度 O(n log n)，空间 O(n)。
```c
void _Merge_sort(int* arr, int left, int right, int* tmp, int n)
{
	//递归
	if (left >= right)
	{
		return;
	}
	//根据mid不断划分为左右序列，[left,mid][mid+1,right]。
	int mid = (right - left) / 2 + left;
	_Merge_sort(arr, left, mid, tmp, n);
	_Merge_sort(arr, mid + 1, right, tmp, n);

	//使区间有序,注意这里的index不能等于0，哪个小哪个往前放
	//这里我在自己写的时候遇到了一个bug，tmp[begin1++] = arr[index++];，这条语句是错的，应该是把原数组里面begin的位置放到tmp那里
	int begin1 = left, end1 = mid;
	int begin2 = mid + 1, end2 = right;
	int index = begin1;
	while (begin1 <= end1 && begin2 <= end2)
	{
		if (arr[begin1] <= arr[begin2])
		{
			tmp[index++] = arr[begin1++];
		}
		else
		{
			tmp[index++] = arr[begin2++];
		}
	}
	//此时还有一个可能没有完全插入
	while (begin1 <= end1)
	{
		tmp[index++] = arr[begin1++];
	}
	while (begin2 <= end2)
	{
		tmp[index++] = arr[begin2++];
	}

	for (int i = left;i <= right;i++)
	{
		arr[i] = tmp[i];
	}
}

void Merge_sort(int* arr, int n)
{
	int* tmp = (int*)malloc(n * sizeof(int));
	if (tmp == NULL)
	{
		perror("Malloc Failed");
		return;
	}
	_Merge_sort(arr, 0, n - 1, tmp, n);
	free(tmp);
}
```
## 归并排序之外部排序
> 假如说我们目前有上亿个待排序数据，如果我们再次采用之前的排序方式，难道要定义一个有上亿个空间的数组吗？这显而易见是不合理的，因此我们要学会使用所谓的外部排序，使用文件来控制大量的数据

### 整体框架
1. 我们要创造出对应的数据并且写入文件中。
2. 我们从创建的文件中依次读取n个数据并且排列有序，即文件file1和file2
3.  之后我们用归并的思想将file1和file2归并为mfile，然后将mfile重命名为file1，再次读取n个数据放入file2中，不断进行循环，直到file2读取的元素个数为0。
因此我们可以写出main主函数：
```c
int main()
{
	Creat_number();
	int number = 200;
	const char* file1 = "file1.txt";
	const char* file2 = "file2.txt";
	const char* mfile = "mfile.txt";
	FILE* fout = fopen("data.txt", "r");
	if (fout == NULL)
	{
		perror("Open Failed");
		return;
	}
	ReadNumSort(fout, number, file1);
	ReadNumSort(fout, number, file2);

	while (1)
	{
		MergeFile(file1, file2, mfile);
		remove(file1);
		remove(file2);
		rename(mfile, file1);
		if (ReadNumSort(fout, number, file2) == 0)
		{
			break;
		}
	}
	fclose(fout);
	return 0;
}
```
其中Creat_number函数是创造数据，ReadNumSort函数实现的是从主文件中读取n个元素并且放入对应文件中，MergeFile函数是将两个文件合并到mfile中去。
### Creat_number函数
```c
void Creat_number()
{
	int n = 1000;
	srand((unsigned int)time(0));
	const char* file = "data.txt";
	FILE* fout = fopen(file, "w");
	if (fout == NULL)
	{
		perror("Create Failed");
		return;
	}
	for (int i = 0;i < n;i++)
	{
		int x = rand() + i;
		fprintf(fout, "%d\n", x);
	}
	fclose(fout);
}
```
### ReadNumSort函数
我们整体分为以下几步，我们首先开辟一个n个元素的数组（方便之后排序使用），然后从主文件中读取**待定个数据**（注意这里不一定是读取n个数据，循环到后期，主文件剩余元素个数可能小于n），之后对数组元素进行排序，再次将有序元素写回对应文件中：
```c
int compare(const void* elem1, const void* elem2)
{
	return *(int*)elem1 - *(int*)elem2;
}
//从文件中读取n个数据并且排列成有序
int ReadNumSort(FILE* fout, int n, const char* file)
{
	//动态开辟空间排序
	int* arr = (int*)malloc(n * sizeof(int));
	int j = 0;
	int x = 0;
	if (arr == NULL)
	{
		perror("Malloc Failded");
		return 0;
	}
	for (int i = 0;i < n;i++)
	{
		if (fscanf(fout, "%d", &x) == EOF)
		{
			break;
		}
		arr[j++] = x;
	}
	qsort(arr, j, sizeof(int), compare);

	//将有序的数组写入file中
	FILE* fin = fopen(file, "w");
	if (fin == NULL)
	{
		perror("Open Failed");
		free(arr);
		return 0;
	}
	for (int i = 0;i < j;i++)
	{
		fprintf(fin, "%d\n", arr[i]);
	}
	free(arr);
	fclose(fin);
	return j;
}
```
### MergeFile函数
这个函数的整体思路为，首先为以只读的方式打开file1和file2两个文件（**以"w"的形式打开会覆盖原有数据**），然后比较file1和file2对应元素大小，小的插入到mfile文件中，循环结束后再将file1或file2的剩余元素插入即可：
```c
void MergeFile(const char* file1, const char* file2, const char* mfile)
{
	//打开三个文件
	FILE* fout1 = fopen(file1, "r");
	FILE* fout2 = fopen(file2, "r");
	FILE* mfout = fopen(mfile, "w");
	if (fout1 == NULL)
	{
		perror("Open Failed");
		return;
	}
	if (fout2 == NULL)
	{
		perror("Open Failed");
		return;
	}
	if (mfout == NULL)
	{
		perror("Open Failed");
		return;
	}
	int x1 = 0;
	int x2 = 0;
	int ret1 = fscanf(fout1, "%d", &x1);
	int ret2 = fscanf(fout2, "%d", &x2);
	while (ret1 != EOF && ret2 != EOF)
	{
		if (x1 < x2)
		{
			fprintf(mfout, "%d\n", x1);
			ret1 = fscanf(fout1, "%d", &x1);
		}
		else
		{
			fprintf(mfout, "%d\n", x2);
			ret2 = fscanf(fout2, "%d", &x2);
		}
	}
	while (ret1 != EOF)
	{
		fprintf(mfout, "%d\n", x1);
		ret1 = fscanf(fout1, "%d", &x1);
	}
	while (ret2 != EOF)
	{
		fprintf(mfout, "%d\n", x2);
		ret2 = fscanf(fout2, "%d", &x2);
	}
	fclose(fout1);
	fclose(fout2);
	fclose(mfout);
}
```
