* ```c
	#define _CRT_SECURE_NO_WARNINGS
	#include <stdio.h>
	#include <stdlib.h>
	void Rotate(int nums[], int k);
	int main()
	{
		int n = 0;
		scanf("%d", &n);
		int* nums = (int*)malloc(n * (sizeof(int)));
		for (int i = 0;i < n;i++)
		{
			scanf("%d", &nums[i]);
		}
		int k = 0;
		scanf("%d", &k);
		Rotate(nums, k);
		free(nums);
		return 0;
	}
	void Rotate(int nums[], int k)
	{
		int size_nums = sizeof(nums) / sizeof(nums[0]);
		while (k--)
		{
			int temp = nums[size_nums - 1];
			for (int i = 0;i < size_nums - 1;i--)
			{
				nums[i] = nums[i - 1];
			}
			nums[0] = temp;
		}
	}
	双重循环实现，但是时间复杂度为O(N)，空间复杂度为O(1);

* ```c
  前n-k个元素逆转，后k个元素逆转，最后整体n个元素逆转，是代码实现的最优解，空间时间利用率都是最高的。
  void Reverse(int *nums,int n,int m)
{
    while(n<m)
    {
        int tem=nums[n];
        nums[n]=nums[m];
        nums[m]=tem;
        n++;
        m--;
    }
}
void rotate(int* nums, int numsSize, int k)
{
    k=k%numsSize;//防止越界情况
    Reverse(nums,0,numsSize-k-1);
    Reverse(nums,numsSize-k,numsSize-1);
    Reverse(nums,0,numsSize-1);
}

void Rotate3(int nums[], int k, int n)
{
	k = k % n;
	int* tem = (int*)malloc(n * sizeof(int));
	for (int i = 0;i < n;i++)
	{
		tem[(i + 2) % n] = nums[i];
	}
	for (int i = 0;i < n;i++)
	{
		nums[i] = tem[i];
	}

}
